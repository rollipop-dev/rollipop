---
title: Bundler API
---

The Bundler API provides programmatic access to Rollipop's bundling and development server functionality. This is useful for integrating Rollipop into custom build tools or workflows.

## `loadConfig`

Load and resolve the Rollipop configuration file.

```ts
function loadConfig(options?: LoadConfigOptions): Promise<ResolvedConfig>;
```

### Parameters

| Parameter            | Type                            | Description                                          |
| -------------------- | ------------------------------- | ---------------------------------------------------- |
| `options.cwd`        | `string`                        | Current working directory (default: `process.cwd()`) |
| `options.configFile` | `string`                        | Path to the configuration file                       |
| `options.mode`       | `'development' \| 'production'` | Mode to run the bundler in                           |
| `options.context`    | `{ command?: string }`          | Additional context to pass to a config function      |

### Returns

Returns a `Promise<ResolvedConfig>` containing the fully resolved configuration.

### Example

```ts
import * as Rollipop from 'rollipop';

const config = await Rollipop.loadConfig();

console.log('Resolved config:', config);
```

## `runServer`

Start the React Native development server.

```ts
function runServer(config: ResolvedConfig, options: ServerOptions): Promise<DevServer>;
```

### Parameters

`config: ResolvedConfig`

The resolved Rollipop configuration (obtained from `loadConfig`).

`options: ServerOptions`

| Option         | Type                          | Description                                                    | Default     |
| -------------- | ----------------------------- | -------------------------------------------------------------- | ----------- |
| `port`         | `number`                      | Port to start the development server on                        | `8081`      |
| `host`         | `string`                      | Host to start the development server on                        | `localhost` |
| `https`        | `boolean`                     | Reserved; `true` currently throws an unsupported-feature error | `false`     |
| `key`          | `string`                      | Reserved; TLS keys are not currently used                      | -           |
| `cert`         | `string`                      | Reserved; TLS certificates are not currently used              | -           |
| `buildOptions` | `Pick<BuildOptions, 'cache'>` | Dev-server build options                                       |             |

Server build options currently expose only `cache`, a boolean that defaults to `true`.

### Returns

Returns a `Promise<DevServer>` containing the running development server instance.

### DevServer Interface

The `DevServer` combines `DevServerContext` with `Emitter<DevServerEvents>` and provides access
to server instances and APIs:

```ts
type DevServer = DevServerContext &
  Emitter<DevServerEvents> & {
    instance: FastifyInstance;
    middlewares: Middlewares;
  };

interface DevServerContext {
  serverBaseUrl: string;
  config: ResolvedConfig;
  options: ServerOptions;
  bundlerPool: BundlerPool;
  eventBus: EventBus;
  state: DevServerState;
  message: ws.Server & {
    broadcast: (method: string, params?: Record<string, any>) => void;
  };
  events: ws.Server & {
    reportEvent: (event: { type: string; [key: string]: unknown }) => void;
  };
  hot: ws.Server & {
    send: (client: ws.WebSocket, eventName: string, payload?: unknown) => void;
    sendAll: (eventName: string, payload?: unknown) => void;
  };
}

type DevServerEvents = {
  'client.connected': { client: WebSocketClient };
  'client.message': { client: WebSocketClient; data: ws.RawData };
  'client.error': { client: WebSocketClient; error: Error };
  'client.disconnected': { client: WebSocketClient };
};
```

### Example

```ts
import * as Rollipop from 'rollipop';

const config = await Rollipop.loadConfig();

const server = await Rollipop.runServer(config, {
  port: 8081,
  host: 'localhost',
});

// Listen to HMR client events
server.on('client.connected', ({ client }) => {
  console.log('Client connected:', client.id);
});

server.on('client.disconnected', ({ client }) => {
  console.log('Client disconnected:', client.id);
});

server.on('client.message', ({ client, data }) => {
  console.log('Message from client:', client.id, data);
});

server.on('client.error', ({ client, error }) => {
  console.error('Client error:', client.id, error);
});

console.log('Dev server running at:', server.instance.listeningOrigin);
```

## `runBuild`

Build a JavaScript bundle for production or development.

```ts
function runBuild(config: ResolvedConfig, options: BuildOptions): Promise<OutputChunk>;
```

### Parameters

`config: ResolvedConfig`

The resolved Rollipop configuration (obtained from `loadConfig`).

`options: BuildOptions`

| Option             | Type                               | Description                                                     | Default                         |
| ------------------ | ---------------------------------- | --------------------------------------------------------------- | ------------------------------- |
| `platform`         | `string`                           | Required target platform, such as `ios` or `android`            | -                               |
| `dev`              | `boolean`                          | Build in development mode                                       | `config.mode === 'development'` |
| `minify`           | `rolldown.OutputOptions['minify']` | Minify the bundle                                               | `config.output.minify ?? false` |
| `cache`            | `boolean`                          | Enable or disable bundler caching                               | `true`                          |
| `outfile`          | `string`                           | Output file path                                                | -                               |
| `sourcemap`        | `boolean \| 'inline' \| 'hidden'`  | Generate a source map                                           | `config.output.sourcemap`       |
| `sourcemapOutfile` | `string`                           | Custom source map path when `sourcemap` is `true` or `'hidden'` | -                               |
| `assetsDir`        | `string`                           | Assets directory path                                           | -                               |

### Returns

Returns a `Promise<OutputChunk>` containing the build result.

### Example

```ts
import * as Rollipop from 'rollipop';

const config = await Rollipop.loadConfig();

const result = await Rollipop.runBuild(config, {
  platform: 'ios',
  dev: false,
  minify: true,
  outfile: './dist/main.bundle.js',
  sourcemap: true,
  sourcemapOutfile: './dist/main.bundle.map',
  assetsDir: './dist/assets',
});

console.log('Bundle built:', result.fileName);
```
